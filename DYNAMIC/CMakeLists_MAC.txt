cmake_minimum_required(VERSION 3.16 FATAL_ERROR)

# ----------------------------------------------------------------------------
# macOS: architecture + deployment target.
# These MUST be set before project() so the compiler checks use them.
# ----------------------------------------------------------------------------
if(CMAKE_HOST_APPLE AND NOT CMAKE_SYSTEM_NAME)
    if(NOT CMAKE_OSX_ARCHITECTURES AND CMAKE_HOST_SYSTEM_PROCESSOR)
        # Match the host by default (Apple Silicon -> arm64)
        set(CMAKE_OSX_ARCHITECTURES "${CMAKE_HOST_SYSTEM_PROCESSOR}" CACHE STRING "Build architectures for macOS")
    endif()
    if(NOT CMAKE_OSX_DEPLOYMENT_TARGET)
        set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum macOS version")
    endif()
endif()

project(NukeCGDenoiser LANGUAGES CXX)

# ============================================================================
# Global settings
# ============================================================================
set(CMAKE_CXX_STANDARD 17)          # DDImage headers do NOT compile as C++20
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE TRUE)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

# Allow global toolchain to be set using GLOBAL_TOOLCHAIN env
if(DEFINED ENV{GLOBAL_TOOLCHAIN})
    message(STATUS "Set toolchain $ENV{GLOBAL_TOOLCHAIN}")
    set(CMAKE_TOOLCHAIN_FILE $ENV{GLOBAL_TOOLCHAIN})
endif()

set(CMAKE_MODULE_PATH "CMake;${CMAKE_MODULE_PATH};${CMAKE_SOURCE_DIR}/cmake/Modules")

if(APPLE)
    message(STATUS "macOS architecture:      ${CMAKE_OSX_ARCHITECTURES}")
    message(STATUS "macOS deployment target: ${CMAKE_OSX_DEPLOYMENT_TARGET}")
endif()

# ============================================================================
# Nuke Configuration
# ============================================================================
if(NOT DEFINED NUKE_VERSION)
    set(NUKE_VERSION "17.0v1")
endif()

# Plugin file extension expected by Nuke on this platform
if(APPLE)
    set(NUKE_PLUGIN_SUFFIX ".dylib")
elseif(WIN32)
    set(NUKE_PLUGIN_SUFFIX ".dll")
else()
    set(NUKE_PLUGIN_SUFFIX ".so")
endif()

# NUKE_INSTALL_DIR may be passed explicitly; otherwise guess per platform.
if(NOT DEFINED NUKE_INSTALL_DIR)
    if(APPLE)
        # Nuke ships as an .app bundle; the SDK lives in Contents/MacOS
        set(_nuke_candidates
            "/Applications/Nuke${NUKE_VERSION}/Nuke${NUKE_VERSION}.app/Contents/MacOS"
            "/Applications/Nuke${NUKE_VERSION}/NukeX${NUKE_VERSION}.app/Contents/MacOS"
            "/Applications/Nuke${NUKE_VERSION}/NukeStudio${NUKE_VERSION}.app/Contents/MacOS"
        )
    elseif(WIN32)
        set(_nuke_candidates
            "C:/Program Files/Nuke${NUKE_VERSION}"
        )
    else()
        set(_nuke_candidates
            "/opt/Nuke${NUKE_VERSION}"
            "/usr/local/Nuke${NUKE_VERSION}"
        )
    endif()

    foreach(_cand IN LISTS _nuke_candidates)
        if(EXISTS "${_cand}/include/DDImage/Iop.h")
            set(NUKE_INSTALL_DIR "${_cand}")
            break()
        endif()
    endforeach()

    if(NOT DEFINED NUKE_INSTALL_DIR)
        message(FATAL_ERROR
            "Could not locate the Nuke ${NUKE_VERSION} SDK.\n"
            "Tried: ${_nuke_candidates}\n"
            "Pass it explicitly, e.g.\n"
            "  cmake -DNUKE_INSTALL_DIR=/Applications/Nuke${NUKE_VERSION}/Nuke${NUKE_VERSION}.app/Contents/MacOS ..")
    endif()
endif()

set(NUKE_INCLUDE_DIR "${NUKE_INSTALL_DIR}/include")
if(NOT EXISTS "${NUKE_INCLUDE_DIR}/DDImage/Iop.h")
    message(FATAL_ERROR "DDImage headers not found in: ${NUKE_INCLUDE_DIR}")
endif()
message(STATUS "Nuke install directory: ${NUKE_INSTALL_DIR}")
message(STATUS "Nuke include directory: ${NUKE_INCLUDE_DIR}")

find_library(NUKE_LIBRARIES
    NAMES DDImage libDDImage
    PATHS "${NUKE_INSTALL_DIR}"
    NO_DEFAULT_PATH
    REQUIRED
)
message(STATUS "Nuke Libraries: ${NUKE_LIBRARIES}")

# ============================================================================
# OpenImageDenoise Configuration (Dynamic - GPU + CPU)
# ============================================================================
if(NOT DEFINED OIDN_ROOT)
    if(DEFINED ENV{OIDN_ROOT})
        set(OIDN_ROOT $ENV{OIDN_ROOT})
    else()
        message(FATAL_ERROR "OIDN_ROOT not set. Please specify: cmake -DOIDN_ROOT=/path/to/oidn ..")
    endif()
endif()

message(STATUS "OIDN_ROOT: ${OIDN_ROOT}")

# Determine lib directory
if(EXISTS "${OIDN_ROOT}/lib64")
    set(OIDN_LIB_DIR "${OIDN_ROOT}/lib64")
elseif(EXISTS "${OIDN_ROOT}/lib")
    set(OIDN_LIB_DIR "${OIDN_ROOT}/lib")
else()
    message(FATAL_ERROR "Could not find OIDN library directory in ${OIDN_ROOT}")
endif()

find_path(OIDN_INCLUDE_DIR
    NAMES OpenImageDenoise/oidn.hpp
    PATHS ${OIDN_ROOT}/include
    NO_DEFAULT_PATH
    REQUIRED
)

# Find the SHARED library (.dylib / .so / .lib)
find_library(OIDN_SHARED_LIB
    NAMES OpenImageDenoise
    PATHS ${OIDN_LIB_DIR}
    NO_DEFAULT_PATH
    REQUIRED
)

message(STATUS "========================================")
message(STATUS "OIDN Configuration (DYNAMIC - GPU+CPU):")
message(STATUS "  Include:    ${OIDN_INCLUDE_DIR}")
message(STATUS "  Shared lib: ${OIDN_SHARED_LIB}")
message(STATUS "  Lib dir:    ${OIDN_LIB_DIR}")
message(STATUS "========================================")

# ============================================================================
# RPATH Configuration - find OIDN (and DDImage) shared libs at runtime
# ============================================================================
set(CMAKE_SKIP_BUILD_RPATH FALSE)
set(CMAKE_BUILD_WITH_INSTALL_RPATH TRUE)
set(CMAKE_INSTALL_RPATH_USE_LINK_PATH TRUE)

if(APPLE)
    set(CMAKE_MACOSX_RPATH TRUE)
    # libDDImage.dylib has an @rpath install name, so the Nuke dir must be on
    # the plugin's rpath list too. @loader_path entries let you ship the OIDN
    # dylibs next to the plugin later without relinking.
    set(DENOISER_RPATHS
        "${OIDN_LIB_DIR}"
        "${NUKE_INSTALL_DIR}"
        "@loader_path"
        "@loader_path/../lib"
    )
else()
    set(DENOISER_RPATHS
        "${OIDN_LIB_DIR}"
        "$ORIGIN"
        "$ORIGIN/../lib"
    )
endif()

# ============================================================================
# Build Denoiser Plugin
# ============================================================================
set(DENOISER_SRC "${CMAKE_SOURCE_DIR}/src/denoiser.cpp")
if(NOT EXISTS ${DENOISER_SRC})
    message(FATAL_ERROR "denoiser.cpp not found at ${DENOISER_SRC}")
endif()
message(STATUS "Found denoiser.cpp at ${DENOISER_SRC}")

set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)

add_library(Denoiser SHARED ${DENOISER_SRC})

set_target_properties(Denoiser PROPERTIES
    PREFIX ""
    SUFFIX "${NUKE_PLUGIN_SUFFIX}"
    CXX_STANDARD 17
    CXX_STANDARD_REQUIRED ON
    BUILD_RPATH   "${DENOISER_RPATHS}"
    INSTALL_RPATH "${DENOISER_RPATHS}"
)

target_include_directories(Denoiser PRIVATE
    ${NUKE_INCLUDE_DIR}
    ${OIDN_INCLUDE_DIR}
)

# Dynamic linking - just link the main shared library.
# GPU device libs (Metal/CUDA/SYCL/HIP) are loaded at runtime by OIDN itself.
target_link_libraries(Denoiser PRIVATE
    ${NUKE_LIBRARIES}
    ${OIDN_SHARED_LIB}
    ${CMAKE_DL_LIBS}
)

# ============================================================================
# Install
# ============================================================================
if(CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT)
    if(WIN32)
        set(CMAKE_INSTALL_PREFIX "$ENV{USERPROFILE}/.nuke" CACHE PATH "" FORCE)
    else()
        set(CMAKE_INSTALL_PREFIX "$ENV{HOME}/.nuke" CACHE PATH "" FORCE)
    endif()
endif()

install(TARGETS Denoiser DESTINATION ".")

message(STATUS "========================================")
message(STATUS "Build Configuration Summary:")
message(STATUS "  Plugin:         Denoiser${NUKE_PLUGIN_SUFFIX}")
message(STATUS "  Platform:       ${CMAKE_SYSTEM_NAME}")
message(STATUS "  Nuke Version:   ${NUKE_VERSION}")
message(STATUS "  Nuke SDK:       ${NUKE_INSTALL_DIR}")
message(STATUS "  OIDN:           DYNAMIC (GPU + CPU)")
message(STATUS "  RPATH:          ${DENOISER_RPATHS}")
message(STATUS "  Output:         ${CMAKE_LIBRARY_OUTPUT_DIRECTORY}")
message(STATUS "  Install prefix: ${CMAKE_INSTALL_PREFIX}")
message(STATUS "========================================")
